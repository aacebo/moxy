# Known differences from syn, one `TAG kind: source` per line, under a comment
# that says why. Each reason is checked against rustc's parser: a source in an
# item under `#[cfg(any())]` parses or not without the later checks. The corpus
# test fails when one of these no longer differs, so the list only shrinks.

# Intentional: nightly or unstable syntax, out of scope
ACCEPT b: { let x = static || 1; }
ACCEPT e: static move || 1
ACCEPT e: static || 1
REJECT e: become f()
REJECT f: const X: u8 = 1 where T: U;
REJECT f: const trait A {}
REJECT f: struct S { a: u8 = 1 }
REJECT t: impl const A
REJECT t: unsafe<'a> &'a A

# Intentional: bare trait objects with parenthesized arguments, which rustc's
# parser accepts and newer editions reject later; syn rejects them
ACCEPT e: <a::Fn(A) -> B>::c
ACCEPT e: f::<Fn::(A) -> B>()
ACCEPT t: A<a::b()>
ACCEPT t: Box<Fn(A) -> B>
ACCEPT t: F::<A>::Fn::(A) -> B
ACCEPT t: Fn()
ACCEPT t: Fn() -> ()
ACCEPT t: Fn(A) -> B
ACCEPT t: Fn(A) -> B + C
ACCEPT t: Fn(A) -> B + Send
ACCEPT t: Fn(A) -> Fn(B) -> C
ACCEPT t: Fn(A) -> impl C
ACCEPT t: Fn(A, B)
ACCEPT t: Fn(A, B) -> C
ACCEPT t: Fn::(A) -> B
ACCEPT t: FnMut(&mut A) -> B<C>
ACCEPT t: FnOnce(&mut A) -> &B
ACCEPT t: a::Fn(A)
ACCEPT t: a::Fn(A) -> B

# Intentional: edition 2015 parameters without a name in a trait, which syn
# rejects
ACCEPT f: trait A { fn f(&u8, mut x: u8); }
ACCEPT f: trait A { fn f(u8); }

# Intentional: edition 2015 `try!`, which moxy lexes as the keyword `try` of
# later editions
REJECT b: { try!(x) }
REJECT e: try!(a)
REJECT f: macro_rules! try { () => {} }

# Intentional: rustc's parser accepts these and rejects them later, where syn
# rejects them or keeps them as tokens
ACCEPT e: A::<B>!()
ACCEPT e: a.1e1
ACCEPT f: fn f() { x.0; x.0.1; x.0 .1; x.0. 1; (x.0).1; x.00; x.0x0; x.1e1; x.1_0; }
ACCEPT f: impl A { type T where Self: Sized = u8; }
ACCEPT f: safe fn f() {}
ACCEPT f: safe static X: u8 = 1;
ACCEPT f: unsafe static X: u8 = 1;
ACCEPT f: use Self::A;
ACCEPT f: use a::Self;
ACCEPT f: use a::{::b};
ACCEPT p: -'a'
ACCEPT t: dyn
ACCEPT t: impl
ACCEPT p: [1.., 2]
ACCEPT p: [1..]
TREE f: trait A { pub fn f(); }

# Intentional: rustc's parser accepts parenthesized arguments on an associated
# item's name and a variadic before another parameter, and rejects them later;
# syn rejects them
ACCEPT f: extern "C" { fn f(..., ...); }
ACCEPT f: extern "C" { fn f(a: u8, ..., b: u8); }
ACCEPT f: fn f(a: u8, x: ..., b: u8) {}
ACCEPT t: A<Item(T) -> U: Clone>
ACCEPT t: A<Item(T) = u8>
ACCEPT t: A<Item(T): Clone>
ACCEPT t: A<Item::(T): Clone>
ACCEPT t: fn(..., ...,)
ACCEPT t: fn(u8, ..., u8)

# Intentional: rustc rejects these while parsing, as moxy does, and syn accepts
# them: an obsolete macro call with an identifier, a static without a type, a
# keyword as a lifetime name, and `mut self` as a pattern
REJECT f: m! ident ();
REJECT f: m! ident {}
REJECT f: static X = 1;
REJECT f: type Symbols<'false> = Table<Symbol<'false>>;
REJECT p: mut self

# Intentional: rustc's lexer rejects an exponent without digits, even in macro
# input, which syn accepts as a suffix
REJECT e: 1e
REJECT f: fn f() { 1f16; 1.0f128; 1f16; 2b; 1px; 1em; 0x1p3; 1_; 1__; }

# Intentional: moxy keeps tokens as written where syn normalizes them: the
# obsolete `...` range, the text of a tuple index, generic parameter order
TOKENS t: A<T, 'a>
TREE e: a.01
TREE e: a.0_0
TREE e: a.0x1
TREE e: match a { 0...5 => 1 }
TREE p: 1...2

# Intentional: moxy models these differently from syn: `self` as a path pattern,
# a block statement and a following `;` as two statements, and attributes before
# a range on its start
TREE b: { #[a] 1..2; }
TREE b: { {}; -1 }
TREE p: self

# Intentional: moxy emits the empty `<>`, `:`, and `where` and the variant
# visibility that the source has, which syn drops
TOKENS f: enum E { pub A }
TOKENS f: fn f() where {}
TOKENS f: fn f<>() {}
TOKENS f: fn min<T:>(v1: T) -> T {}
TOKENS f: impl<> Bound<> {}
TOKENS f: impl<> X {}
TOKENS f: struct S<>;
TOKENS f: struct S<T: >(T);
TOKENS f: trait A: {}
TOKENS i: struct S<T: >(T);

# Intentional: moxy keeps a shebang of a path and one word as the file's
# shebang tokens; syn drops the line
TOKENS f: #!/usr/bin/env rust\n fn f() {}
TOKENS f: #!/usr/bin/env rust\n#![allow(a)]\nfn main() {}
