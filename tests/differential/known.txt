# Known differences from syn, one `TAG kind: source` per line, under a comment that
# says why. The corpus test fails when one of these no longer differs, so the list
# only shrinks.

# Intentional: inner attributes attach to the first item or statement of their
# container, not to the container as in syn; a container without items has no
# place for them, so a lone inner attribute is rejected
REJECT f: extern "C" { #![a] }
REJECT f: fn f() { #![a] }
REJECT f: fn f() { //! inner\n }
REJECT f: fn f() { loop { #![a] } }
REJECT f: fn f() { match x { #![a] } }
REJECT f: impl A { #![a] }
REJECT f: mod m { #![a] }
REJECT f: mod m { //! doc\n }
REJECT f: trait A { #![a] }
REJECT i: fn f() { #![allow(x)] }
REJECT i: mod m { #![allow(x)] }
TREE f: fn f() { #![a] let x = 1; }
TREE f: fn f() { #![a] x }
TREE f: fn f() { #![allow(unused)] x; y }
TREE f: fn f() { //! doc\n x }
TREE f: fn f() { for x in y { #![a] z; } }
TREE f: fn f() { match x { #![a] _ => 1 } }
TREE f: fn f() { unsafe { #![a] x } }
TREE f: fn f() { while x { #![a] y; } }
TREE f: fn f() { { #![a] x } }
TREE f: impl A { #![inner] fn f() {} }
TREE f: mod m { #![a] #![b] use c; }
TREE f: mod m { #![a] fn f() {} }
TREE f: mod m { //! doc\n fn f() {} }
TREE f: trait A { #![a] fn f(); }

ACCEPT t: Fn()

ACCEPT t: Fn(A) -> B
ACCEPT t: Fn(A) -> B + C
ACCEPT t: Fn(A) -> B + Send
ACCEPT t: Fn(A) -> Fn(B) -> C
ACCEPT t: Fn(A) -> impl C
ACCEPT t: Fn(A, B)
ACCEPT t: Fn(A, B) -> C
ACCEPT t: Fn::(A) -> B
ACCEPT t: FnMut(&mut A) -> B<C>

# Intentional: rustc's parser accepts these and rejects them later, which syn
# rejects while parsing or keeps as tokens

TREE f: trait A { pub fn f(); }

# Intentional: a type alias's where clause before `=`, the older place, emits
# after the type, where Rust prefers it, as for associated types
TOKENS f: type T<U> where U: A = Vec<U>;

# Intentional: moxy keeps tokens as written where syn normalizes them: the
# obsolete `...` range, a leading zero in a tuple index, generic parameter order
TOKENS t: A<T, 'a>
TREE e: a.01
TREE e: a.0_0
TREE e: a.0x1
TREE e: match a { 0...5 => 1 }
TREE p: 1...2

# Intentional: moxy models these differently from syn: `self` as a path pattern,
# a block statement and a following `;` as two statements, and attributes before
# a range on its start
TREE b: { #[a] 1..2; }
TREE b: { {}; -1 }
TREE p: self

# Intentional: moxy emits the empty `<>`, `:`, and `where` and the variant

TOKENS f: impl<> Bound<> {}
TOKENS f: impl<> X {}
TOKENS f: struct S<T: >(T);
TOKENS f: struct S<>;
TOKENS f: fn f<>() {}
TOKENS i: struct S<T: >(T);
TOKENS f: enum E { pub A }
TOKENS f: fn f() where {}
TOKENS f: trait A: {}

# Intentional: moxy keeps a shebang of a path and one word as the file's
# shebang tokens, which syn drops from the tokens
TOKENS f: #!/usr/bin/env rust\n fn f() {}
TOKENS f: #!/usr/bin/env rust\n#![allow(a)]\nfn main() {}

# Intentional: rustc's lexer rejects an exponent without digits, even in macro
# input, which syn accepts as a suffix
REJECT e: 1e
REJECT f: fn f() { 1f16; 1.0f128; 1f16; 2b; 1px; 1em; 0x1p3; 1_; 1__; }

# Intentional: rustc rejects these after parsing, as moxy does while parsing:

REJECT f: type Symbols<'false> = Table<Symbol<'false>>;
REJECT f: impl<T> X<T> { type Item; #[inline] fn next() -> Option<T> {} fn advance_by(self, n: usize) -> Result<(), NonZero<usize>> { rem = if len < n {} else {} } }
